include(power_blue_roboticsTaskLib)
ADD_LIBRARY(${POWER_BLUE_ROBOTICS_TASKLIB_NAME} SHARED
    ${POWER_BLUE_ROBOTICS_TASKLIB_SOURCES})
add_dependencies(${POWER_BLUE_ROBOTICS_TASKLIB_NAME}
    regen-typekit)


TARGET_LINK_LIBRARIES(${POWER_BLUE_ROBOTICS_TASKLIB_NAME}
    ${OrocosRTT_LIBRARIES}
    ${QT_LIBRARIES}
    ${POWER_BLUE_ROBOTICS_TASKLIB_DEPENDENT_LIBRARIES})
SET_TARGET_PROPERTIES(${POWER_BLUE_ROBOTICS_TASKLIB_NAME}
    PROPERTIES LINK_INTERFACE_LIBRARIES "${POWER_BLUE_ROBOTICS_TASKLIB_INTERFACE_LIBRARIES}")
SET_TARGET_PROPERTIES(${POWER_BLUE_ROBOTICS_TASKLIB_NAME}
    PROPERTIES INTERFACE_LINK_LIBRARIES "${POWER_BLUE_ROBOTICS_TASKLIB_INTERFACE_LIBRARIES}")

orogen_apply_task_library_options(${POWER_BLUE_ROBOTICS_TASKLIB_NAME})

INSTALL(TARGETS ${POWER_BLUE_ROBOTICS_TASKLIB_NAME}
    RUNTIME DESTINATION bin
    LIBRARY DESTINATION lib/orocos)

INSTALL(FILES ${POWER_BLUE_ROBOTICS_TASKLIB_HEADERS}
    DESTINATION include/orocos/power_blue_robotics)
